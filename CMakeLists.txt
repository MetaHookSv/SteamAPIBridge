cmake_minimum_required(VERSION 3.21)
option(FORMAT_VALIDATION_ONLY "Configure only clang-format targets without native dependencies" OFF)
if(FORMAT_VALIDATION_ONLY)
    project(FormatValidation LANGUAGES NONE)
    include(cmake/FormatValidation.cmake)
    return()
endif()
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(SteamAPIBridge VERSION 1.0.0 LANGUAGES CXX)
if(WIN32 AND NOT MSVC)
    message(FATAL_ERROR "SteamAPIBridge requires MSVC on Windows.")
elseif(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "SteamAPIBridge supports Windows and Linux.")
endif()
# A parent project keeps its own configurations.
if(PROJECT_IS_TOP_LEVEL)
    get_property(steambridge_multi_config GLOBAL PROPERTY GENERATOR_IS_MULTI_CONFIG)
    if(steambridge_multi_config)
        set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Supported configurations" FORCE)
    elseif(NOT CMAKE_BUILD_TYPE MATCHES "^(Debug|Release)$")
        message(FATAL_ERROR "Select Debug or Release.")
    endif()
endif()
set(STEAMSDK_SOURCE_PATH "$ENV{STEAMSDK_SOURCE_PATH}" CACHE PATH "Read-only SteamSDK source")
include(cmake/Dependencies.cmake)
steambridge_prepare_dependencies()
include(cmake/SourceGroups.cmake)
add_library(SteamAPIBridge SHARED src/SteamAPIBridge.cpp src/Host.cpp src/Platform.h src/LegacyInterfaces.h)
steambridge_assign_source_groups(SteamAPIBridge)
add_library(SteamAPIBridge::SteamAPIBridge ALIAS SteamAPIBridge)
target_compile_features(SteamAPIBridge PRIVATE cxx_std_20)
target_include_directories(SteamAPIBridge PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include" PRIVATE "${STEAMSDK_SOURCE_PATH}")
set_target_properties(SteamAPIBridge PROPERTIES DEBUG_POSTFIX "")
if(MSVC)
    target_compile_definitions(SteamAPIBridge PRIVATE STEAMAPIBRIDGE_EXPORTS WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
    target_compile_options(SteamAPIBridge PRIVATE /W4 /permissive- /Zi)
    target_link_options(SteamAPIBridge PRIVATE /DEBUG)
else()
    target_compile_definitions(SteamAPIBridge PRIVATE STEAMAPIBRIDGE_EXPORTS)
    target_compile_options(SteamAPIBridge PRIVATE -Wall -Wextra)
    target_link_libraries(SteamAPIBridge PRIVATE ${CMAKE_DL_LIBS})
    # Only the SteamBridge_ C ABI is exported; the version script also hides libstdc++ template instances.
    set_target_properties(SteamAPIBridge PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/SteamAPIBridge.map")
    target_link_options(SteamAPIBridge PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/SteamAPIBridge.map")
endif()
# The steam_api runtime that host mode loads from the bridge's directory.
if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(steambridge_runtime "${STEAMSDK_SOURCE_PATH}/bin/steam_api64.dll")
elseif(WIN32)
    set(steambridge_runtime "${STEAMSDK_SOURCE_PATH}/bin/steam_api.dll")
elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(steambridge_runtime "${STEAMSDK_SOURCE_PATH}/bin/linux64/libsteam_api.so")
else()
    set(steambridge_runtime "${STEAMSDK_SOURCE_PATH}/bin/linux32/libsteam_api.so")
endif()
set(STEAMAPIBRIDGE_RUNTIME "${steambridge_runtime}" CACHE INTERNAL "steam_api runtime for the target architecture")
set(STEAMAPIBRIDGE_INSTALL_DIR "svencoop/metahook/dlls" CACHE STRING "Install directory of the bridge, relative to the install prefix")
option(STEAMAPIBRIDGE_INSTALL_RUNTIME "Install the steam_api runtime beside the bridge for host mode" OFF)
install(TARGETS SteamAPIBridge RUNTIME DESTINATION "${STEAMAPIBRIDGE_INSTALL_DIR}" LIBRARY DESTINATION "${STEAMAPIBRIDGE_INSTALL_DIR}")
if(MSVC)
    install(FILES "$<TARGET_PDB_FILE:SteamAPIBridge>" DESTINATION "${STEAMAPIBRIDGE_INSTALL_DIR}")
endif()
if(STEAMAPIBRIDGE_INSTALL_RUNTIME)
    install(FILES "${STEAMAPIBRIDGE_RUNTIME}" DESTINATION "${STEAMAPIBRIDGE_INSTALL_DIR}")
endif()
option(STEAMAPIBRIDGE_BUILD_TESTS "Build bridge regression tests" ${PROJECT_IS_TOP_LEVEL})
if(STEAMAPIBRIDGE_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()

# Shared formatting targets are explicit; ordinary native builds do not run them.
include(cmake/FormatValidation.cmake)
