cmake_minimum_required(VERSION 3.21)
option(FORMAT_VALIDATION_ONLY "Configure only clang-format targets without native dependencies" OFF)
if(FORMAT_VALIDATION_ONLY)
    project(FormatValidation LANGUAGES NONE)
    include(cmake/FormatValidation.cmake)
    return()
endif()
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(SteamAPIBridge VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "SteamAPIBridge requires Windows MSVC x86 (-A Win32).")
endif()
get_property(steambridge_multi_config GLOBAL PROPERTY GENERATOR_IS_MULTI_CONFIG)
if(steambridge_multi_config)
    set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Supported configurations" FORCE)
elseif(NOT CMAKE_BUILD_TYPE MATCHES "^(Debug|Release)$")
    message(FATAL_ERROR "Select Debug or Release.")
endif()
set(STEAMSDK_SOURCE_PATH "$ENV{STEAMSDK_SOURCE_PATH}" CACHE PATH "Read-only SteamSDK source")
set(VC_LTL_Root "$ENV{VC_LTL_Root}" CACHE PATH "VC-LTL root")
include(cmake/Dependencies.cmake)
steambridge_prepare_dependencies()
include("${VC_LTL_Root}/VC-LTL helper for cmake.cmake")
add_library(SteamAPIBridge SHARED src/SteamAPIBridge.cpp)
add_library(SteamAPIBridge::SteamAPIBridge ALIAS SteamAPIBridge)
target_compile_features(SteamAPIBridge PRIVATE cxx_std_20)
target_compile_definitions(SteamAPIBridge PRIVATE STEAMAPIBRIDGE_EXPORTS WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_include_directories(SteamAPIBridge PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include" PRIVATE "${STEAMSDK_SOURCE_PATH}")
target_compile_options(SteamAPIBridge PRIVATE /W4 /permissive- /Zi)
target_link_options(SteamAPIBridge PRIVATE /DEBUG)
set_target_properties(SteamAPIBridge PROPERTIES DEBUG_POSTFIX "")
install(TARGETS SteamAPIBridge RUNTIME DESTINATION svencoop/metahook/dlls)
install(FILES "$<TARGET_PDB_FILE:SteamAPIBridge>" DESTINATION svencoop/metahook/dlls)
option(STEAMAPIBRIDGE_BUILD_TESTS "Build bridge regression tests" ${PROJECT_IS_TOP_LEVEL})
if(STEAMAPIBRIDGE_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()

# Shared formatting targets are explicit; ordinary native builds do not run them.
include(cmake/FormatValidation.cmake)
