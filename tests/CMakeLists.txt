# Test executables share one directory, and a single target copies the bridge there;
# per-executable copies race in parallel builds.
set(steambridge_test_bin "${CMAKE_CURRENT_BINARY_DIR}/bin/$<CONFIG>")
if(WIN32)
    add_custom_target(SteamBridgeTestRuntime
        COMMAND ${CMAKE_COMMAND} -E make_directory "${steambridge_test_bin}"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:SteamAPIBridge>" "${steambridge_test_bin}/$<TARGET_FILE_NAME:SteamAPIBridge>")
    add_dependencies(SteamBridgeTestRuntime SteamAPIBridge)
endif()
function(steambridge_add_test_executable target)
    add_executable(${target} ${ARGN})
    target_link_libraries(${target} PRIVATE SteamAPIBridge)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_compile_options(${target} PRIVATE ${steambridge_test_options})
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${steambridge_test_bin}")
    if(WIN32)
        add_dependencies(${target} SteamBridgeTestRuntime)
    endif()
endfunction()
